	\section{Zakres i cel projektu (opis założeń funkcjonalnych projektowanej bazy danych)}
	
	\subsection{Wprowadzenie}
	
	Realizowany projekt dotyczy przedsiębiorstwa z branży motoryzacyjnej -- salonu samochodowego. Projekt bazy danych ma na celu wsparcie kompleksowej obsługi wszystkich procesów biznesowych związanych ze sprzedażą, serwisowaniem oraz zarządzaniem pojazdami.
	
	\subsection{Opis działalności biznesowej}
	
	Salon samochodowy zajmuje się:
	\begin{itemize}
		\item Sprzedażą nowych i używanych samochodów osobowych
		\item Organizacją jazd testowych dla klientów
		\item Serwisem i naprawami samochodów
		\item Obsługą zamówień specjalnych
	\end{itemize}
	
	\subsection{Główne procesy biznesowe}
	
	System bazy danych będzie wspierał następujące procesy:
	
	\begin{enumerate}
		\item \textbf{Zarządzanie stanem magazynowym pojazdów} -- śledzenie dostępności, statusu i lokalizacji pojazdów w poszczególnych oddziałach salonu
		\item \textbf{Obsługa klientów indywidualnych i firm} -- rejestracja klientów, prowadzenie historii kontaktów i transakcji
		\item \textbf{Realizacja sprzedaży z różnymi formami płatności} -- kompleksowa obsługa procesu sprzedaży
		\item \textbf{Organizacja i rejestracja jazd testowych} -- rezerwacja i ewidencja jazd próbnych
		\item \textbf{Obsługa serwisowa pojazdów} -- rejestracja zleceń naprawczych, ewidencja kosztów i historii serwisowania
		\item \textbf{Zarządzanie pracownikami (sprzedawcy, serwisanci)} -- ewidencja pracowników wraz z ich specjalizacjami
		\item \textbf{Obsługa zamówień indywidualnych} -- przyjmowanie i realizacja zamówień na pojazdy niedostępne w salonie
		\item \textbf{Wystawianie faktur i dokumentacji sprzedaży} -- generowanie dokumentów transakcji
	\end{enumerate}
	
	\subsection{Cele projektowanej bazy danych}
	
	\begin{itemize}
		\item Centralizacja danych o pojazdach, klientach i transakcjach
		\item Wsparcie procesów decyzyjnych poprzez zapewnienie aktualnych danych
		\item Eliminacja redundancji danych
		\item Zapewnienie integralności i spójności informacji
		\item Umożliwienie generowania raportów sprzedażowych i analitycznych
		\item Ułatwienie obsługi klienta poprzez dostęp do pełnej historii współpracy
	\end{itemize}
	
	\section{Definicja systemu}
	
	\subsection{Perspektywy użytkowników}
	
	System będzie wykorzystywany przez cztery główne grupy użytkowników, z których każda ma określone wymagania funkcjonalne.
	
	\subsubsection{Manager Salonu}
	
	\textbf{Cele:} Zarządzanie oddziałem, nadzór nad sprzedażą i pracownikami
	
	\textbf{Operacje:}
	\begin{itemize}
		\item Przeglądanie raportów sprzedaży
		\item Zarządzanie stanem magazynowym
		\item Ocena wyników pracowników
		\item Analiza rentowności
	\end{itemize}
	
	\subsubsection{Sprzedawca}
	
	\textbf{Cele:} Sprzedaż pojazdów, obsługa klientów
	
	\textbf{Operacje:}
	\begin{itemize}
		\item Rejestracja nowych klientów
		\item Organizacja jazd testowych
		\item Realizacja sprzedaży
		\item Stosowanie rabatów (w ramach limitu)
		\item Wystawianie faktur
	\end{itemize}
	
	\subsubsection{Serwisant}
	
	\textbf{Cele:} Naprawa i konserwacja pojazdów
	
	\textbf{Operacje:}
	\begin{itemize}
		\item Rejestracja zleceń serwisowych
		\item Aktualizacja statusu naprawy
		\item Wprowadzanie kosztów części i robocizny
		\item Zamykanie zleceń
	\end{itemize}
	
	\subsubsection{Klient}
	
	\textbf{Cele:} Zakup pojazdu, serwis
	
	\textbf{Operacje:}
	\begin{itemize}
		\item Przeglądanie dostępnych pojazdów
		\item Rezerwacja jazdy testowej
		\item Składanie zamówienia
		\item Sprawdzanie historii zakupów
		\item Zgłaszanie pojazdu do serwisu
	\end{itemize}
	
	\section{Model konceptualny}
	
	\subsection{Definicja zbiorów encji określonych w projekcie (decyzje projektowe)}
	
	Model konceptualny składa się z ośmiu głównych encji.
	
	\subsubsection{SALON}
	
	Encja główna systemu, reprezentująca oddział salonu samochodowego. Umieszczona w lewym górnym rogu diagramu ER jako punkt wyjściowy modelu.
	
	\textbf{Atrybuty:}
	\begin{itemize}
		\item \textbf{nr\_salonu} (PK) -- sztuczny klucz główny
		\item nazwa -- nazwa oddziału
		\item \textbf{adres} -- atrybut złożony (composite) zawierający: ulica, miasto, kod\_pocztowy
		\item telefon -- numer kontaktowy
		\item email -- adres email oddziału
	\end{itemize}
	
	\subsubsection{POJAZD}
	
	Reprezentuje pojedynczy egzemplarz pojazdu znajdujący się w ofercie salonu.
	
	\textbf{Atrybuty:}
	\begin{itemize}
		\item \textbf{nr\_vin} (PK) -- numer identyfikacyjny pojazdu (naturalny klucz)
		\item rok\_produkcji -- rok wyprodukowania pojazdu
		\item przebieg -- liczba przejechanych kilometrów
		\item kolor -- kolor nadwozia
		\item cena\_katalogowa -- typu Money (cena wywoławcza)
		\item stan -- dziedzina \{Nowy, Uzywany\}
		\item status -- dziedzina \{Dostepny, Sprzedany, Rezerwacja, Serwis\}
		\item data\_przyjecia -- data wpływu pojazdu do salonu
		\item uwagi -- typu VARCHAR(500) dla dodatkowych informacji
	\end{itemize}
	
	\subsubsection{MODEL\_SAMOCHODU}
	
	Reprezentuje typ/model pojazdu oferowany przez producenta.
	
	\textbf{Atrybuty:}
	\begin{itemize}
		\item \textbf{id\_modelu} (PK) -- sztuczny klucz główny
		\item nazwa\_modelu -- komercyjna nazwa modelu
		\item typ\_nadwozia -- dziedzina \{Sedan, SUV, Hatchback, Kombi\}
		\item pojemnosc\_silnika -- pojemność w litrach
		\item moc\_km -- moc silnika w koniach mechanicznych
		\item rodzaj\_paliwa -- dziedzina \{Benzyna, Diesel, Elektryczny, Hybryda\}
		\item liczba\_miejsc -- liczba miejsc siedzących
		\item producent\_nazwa -- nazwa producenta
		\item producent_kraj -- kraj producenta
	\end{itemize}
	
	\subsubsection{KLIENT}
	
	Reprezentuje klienta salonu -- osobę fizyczną lub firmę.
	
	\textbf{Atrybuty:}
	\begin{itemize}
		\item \textbf{nr\_klienta} (PK) -- sztuczny klucz główny
		\item typ\_klienta -- dziedzina \{Indywidualny, Firma\}
		\item \textbf{dane\_osobowe} -- atrybut złożony (composite) dla klientów indywidualnych:
		\begin{itemize}
			\item imie
			\item nazwisko
			\item pesel (klucz kandydujący)
		\end{itemize}
		\item \textbf{dane\_firmowe} -- atrybut złożony (composite) dla firm:
		\begin{itemize}
			\item nazwa\_firmy
			\item nip (klucz kandydujący)
		\end{itemize}
		\item \textbf{adres} -- atrybut złożony (composite)
		\item telefon -- numer kontaktowy (obowiązkowy)
		\item email -- adres email
		\item data\_rejestracji -- data dodania do bazy
	\end{itemize}
	
	\textbf{Klucze kandydujące:}
	\begin{itemize}
		\item pesel -- dla klientów indywidualnych
		\item nip -- dla klientów firmowych
	\end{itemize}
	
	\subsubsection{PRACOWNIK (z hierarchią specjalizacji)}
	
	Reprezentuje pracownika salonu. Jest encją nadrzędną w hierarchii dziedziczenia.
	
	\textbf{Atrybuty wspólne:}
	\begin{itemize}
		\item \textbf{nr\_pracownika} (PK) -- sztuczny klucz główny
		\item imie -- imię pracownika
		\item nazwisko -- nazwisko pracownika
		\item pesel -- PESEL (klucz kandydujący)
		\item stanowisko -- zajmowane stanowisko
		\item data\_zatrudnienia -- data rozpoczęcia pracy
		\item pensja\_podstawowa -- wynagrodzenie zasadnicze (typu Money)
	\end{itemize}
	
	\textbf{Specjalizacja: SPRZEDAWCA}
	
	\begin{itemize}
		\item prowizja\_procent -- procent prowizji od sprzedaży
		\item limit\_rabatu\_procent -- maksymalny rabat jaki może przyznać
		\item liczba\_sprzedazy -- licznik sprzedanych pojazdów w bieżącym miesiącu
	\end{itemize}
	
	\textbf{Specjalizacja: SERWISANT}
	
	Atrybuty specyficzne:
	\begin{itemize}
		\item specjalizacja -- obszar specjalizacji technicznej
		\item certyfikaty -- atrybut wielowartościowy (multivalued) lub TEXT zawierający posiadane certyfikaty
		\item stawka\_godzinowa -- stawka za godzinę pracy (typu Money)
	\end{itemize}
	
	\textbf{Właściwości hierarchii:}
	\begin{itemize}
		\item Typ: Overlapping -- pracownik może być jednocześnie sprzedawcą i serwisantem
		\item Uczestnictwo: Partial -- nie każdy pracownik musi należeć do specjalizacji
		\item W encjach specjalizacji nie definiujemy osobnych kluczy głównych (dziedziczą nr\_pracownika)
	\end{itemize}
	
	\subsubsection{SPRZEDAZ}
	
	Reprezentuje związek sprzedaży łączący klienta i pojazd.
	
	\textbf{Atrybuty:}
	\begin{itemize}
		\item \textbf{nr\_sprzedazy} (PK) -- sztuczny klucz główny
		\item data\_sprzedazy -- data zawarcia transakcji
		\item cena\_sprzedazy -- cena wywoławcza (typu Money)
		\item cena\_koncowa -- ostateczna cena po rabacie (typu Money)
		\item metoda\_platnosci -- dziedzina \{Gotowka, Przelew, Finansowanie, Leasing\}
		\item status -- dziedzina \{W\_trakcie, Zakonczona, Anulowana\}
	\end{itemize}
	
	\subsubsection{WYPOSAZENIE}
	
	Reprezentuje element wyposażenia dodatkowego pojazdu.
	
	\textbf{Atrybuty:}
	\begin{itemize}
		\item \textbf{id\_wyposazenia} (PK) -- sztuczny klucz główny
		\item nazwa -- nazwa elementu wyposażenia (klucz kandydujący)
		\item kategoria -- dziedzina \{Bezpieczenstwo, Komfort, Multimedia\}
		\item dodatkowa\_cena -- dopłata za element (typu Money)
	\end{itemize}
	
	\subsection{Ustalenie związków między encjami i ich typów}
	
	\subsubsection{Związki typu 1:M}
	
	\textbf{1. SALON [posiada] POJAZD (1:M)}
	
	Krotność: (1,1) : (0,N)
	\begin{itemize}
		\item Każdy pojazd należy do dokładnie jednego salonu (oddziału)
		\item Salon może posiadać wiele pojazdów lub nie posiadać żadnego
	\end{itemize}
	
	\textbf{2. SALON [zatrudnia] PRACOWNIK (1:M)}
	
	Krotność: (1,1) : (0,N)
	\begin{itemize}
		\item Każdy pracownik jest przypisany do dokładnie jednego salonu
		\item Salon może zatrudniać wielu pracowników
	\end{itemize}
	
	\textbf{3. PRODUCENT [produkuje] MODEL\_SAMOCHODU (1:M)}
	
	Krotność: (1,1) : (1,N)
	\begin{itemize}
		\item Każdy model jest produkowany przez dokładnie jednego producenta
		\item Producent musi mieć przynajmniej jeden model w ofercie
	\end{itemize}
	
	\textbf{4. MODEL\_SAMOCHODU [jest\_instancja] POJAZD (1:M)}
	
	Krotność: (1,1) : (1,N)
	\begin{itemize}
		\item Każdy pojazd jest egzemplarzem dokładnie jednego modelu
		\item Model musi mieć przynajmniej jedną instancję w systemie
	\end{itemize}
	
	\textbf{5. KLIENT [dokonuje] SPRZEDAZ (1:M)}
	
	Krotność: (1,1) : (0,N)
	\begin{itemize}
		\item Każda sprzedaż dotyczy dokładnie jednego klienta
		\item Klient może dokonać wielu zakupów lub żadnego
	\end{itemize}
	
	\textbf{6. PRACOWNIK [realizuje] SPRZEDAZ (1:M)}
	
	Krotność: (1,1) : (0,N)
	\begin{itemize}
		\item Każda sprzedaż jest realizowana przez dokładnie jednego pracownika
		\item Pracownik może zrealizować wiele sprzedaży
	\end{itemize}
	
	\textbf{7. POJAZD [jest\_przedmiotem] SPRZEDAZ (M:1)}
	
	Krotność: (1,1) : (0,1)
	\begin{itemize}
		\item Każda sprzedaż dotyczy dokładnie jednego pojazdu
		\item Pojazd może być przedmiotem maksymalnie jednej sprzedaży (nie można sprzedać tego samego pojazdu dwa razy)
	\end{itemize}
	
	\subsubsection{Związki typu M:N}
	
	\textbf{1. POJAZD [ma\_wyposazenie] WYPOSAŻENIE (M:N)}
	
	Krotność: (1,N) : (0,N)
	\begin{itemize}
		\item Jeden pojazd musi mieć przynajmniej jedno wyposażenie
		\item Jedno wyposażenie może występować w wielu pojazdach
		\item Element wyposażenia może nie być przypisany do żadnego pojazdu (jeśli nie jest obecnie używany)
	\end{itemize}
	
	\textbf{2. KLIENT [rezerwuje] POJAZD poprzez TEST\_DRIVE (M:N)}
	
	Krotność: (0,N) : (0,N)
	\begin{itemize}
		\item Klient może zarezerwować wiele jazd testowych (dla różnych pojazdów)
		\item Pojazd może być użyty w wielu jazdach testowych (przez różnych klientów)
		\item Zarówno klient jak i pojazd mogą nie mieć żadnych jazd testowych
	\end{itemize}
	
	\subsubsection{Związek specjalizacji (ISA)}
	
	\textbf{PRACOWNIK [ISA] \{SPRZEDAWCA, SERWISANT\}}
	
	Właściwości:
	\begin{itemize}
		\item \textbf{Typ:} Overlapping (nakładające się) -- pracownik może być jednocześnie sprzedawcą i serwisantem
		\item \textbf{Uczestnictwo:} Partial (częściowe) -- nie każdy pracownik musi należeć do specjalizacji
		\item \textbf{Konwersja do modelu logicznego:} Wymaga ustawienia generacji na N tablic (osobne tabele dla każdej specjalizacji)
	\end{itemize}

	\subsection{Dodatkowe reguły integralnościowe}
	
	\subsubsection{Reguły domenowe}
	
	\begin{enumerate}
		\item PESEL musi mieć dokładnie 11 znaków cyfr: \texttt{CHAR(11)}
		\item NIP musi mieć dokładnie 10 znaków cyfr: \texttt{CHAR(10)}
		\item Cena pojazdu musi być $> 0$
		\item Rok produkcji: $1900 \leq \text{rok\_produkcji} \leq 2025$
		\item Przebieg $\geq 0$
		\item Stan pojazdu: \texttt{IN ('Nowy', 'Uzywany')}
		\item Metoda płatności: \texttt{IN ('Gotowka', 'Przelew', 'Finansowanie', 'Leasing')}
		\item Email musi być unikalny dla KLIENT i PRACOWNIK
	\end{enumerate}
	
	\subsubsection{Reguły encji}
	
	\begin{enumerate}
		\item Każdy POJAZD musi mieć przypisany MODEL\_SAMOCHODU
		\item SPRZEDAZ wymaga KLIENTA i POJAZDU
		\item Cena końcowa $\leq$ cena sprzedaży (rabat)
		\item Data zatrudnienia $\leq$ data zwolnienia (jeśli istnieje)
		\item KLIENT typu "Indywidualny" musi mieć PESEL
		\item KLIENT typu "Firma" musi mieć NIP
	\end{enumerate}
	
	\subsubsection{Reguły referencyjne}
	
	\begin{enumerate}
		\item POJAZD ze statusem "Sprzedany" musi mieć powiązaną SPRZEDAZ
		\item Nie można usunąć KLIENTA, który ma historię SPRZEDAZY
		\item Nie można usunąć POJAZDU, który został sprzedany
		\item TEST\_DRIVE wymaga ważnego numeru prawa jazdy
	\end{enumerate}

	\subsubsection{Zasady wyboru kluczy}

		\item \textbf{Klucze kandydujące (UK):} Identyfikujemy naturalne klucze, które mogą służyć jako alternatywne identyfikatory
	\end{itemize}

	\subsection{Problem pułapek szczelinowych i wachlarzowych -- analiza i przykłady}
	
	\subsubsection{Pułapka wachlarzowa (Fan Trap)}
	
	\textbf{Definicja:} Pułapka wachlarzowa występuje, gdy model reprezentuje związek między encjami w sposób niejednoznaczny, co prowadzi do problemów z interpretacją ścieżki między encjami.
	
	\textbf{Analiza w projekcie:}
	
	Potencjalna pułapka wachlarzowa mogłaby wystąpić w łańcuchu:
	$$\text{SALON} \rightarrow \text{PRACOWNIK} \rightarrow \text{SPRZEDAZ} \leftarrow \text{POJAZD}$$
	
	\textbf{Problem:} Czy można jednoznacznie określić, w którym salonie został sprzedany pojazd?
	
	\textbf{Rozwiązanie zastosowane w projekcie:}
	
	Model jest zaprojektowany tak, że:
	\begin{itemize}
		\item Każdy POJAZD należy do dokładnie jednego SALON (związek 1:M)
		\item Każdy PRACOWNIK należy do dokładnie jednego SALON (związek 1:M)
		\item Każda SPRZEDAZ jest realizowana przez PRACOWNIKA i dotyczy POJAZDU
	\end{itemize}
	
	Dzięki temu można jednoznacznie określić salon sprzedaży poprzez:
	\begin{itemize}
		\item Salon pracownika realizującego sprzedaż
		\item Salon, do którego należał pojazd przed sprzedażą
	\end{itemize}
	
	\textbf{Reguła weryfikacyjna:}
	
	Należy zapewnić, że pojazd i pracownik realizujący sprzedaż należą do tego samego salonu:
	
	\begin{minted}{sql}
	CHECK (POJAZD.nr_salonu = PRACOWNIK.nr_salonu)
	\end{minted}
	
	\subsubsection{Pułapka szczelinowa (Chasm Trap)}
	
	\textbf{Definicja:} Pułapka szczelinowa występuje, gdy brakuje kompletnej ścieżki między encjami w modelu, co uniemożliwia uzyskanie niektórych informacji.
	
	\textbf{Analiza w projekcie:}
	
	Potencjalna pułapka szczelinowa:
	$$\text{KLIENT} \rightarrow \text{SPRZEDAZ} \leftarrow \text{POJAZD} \rightarrow \text{MODEL\_SAMOCHODU}$$
	
	\textbf{Problem:} Czy można znaleźć wszystkie pojazdy danego modelu, które oglądał klient (bez zakupu)?
	
	Odpowiedź: NIE -- jeśli klient nie dokonał zakupu, nie ma ścieżki przez SPRZEDAZ.
	
	\textbf{Rozwiązanie:}
	
	Wprowadzono encję TEST\_DRIVE, która tworzy alternatywną ścieżkę:
	$$\text{KLIENT} \rightarrow \text{TEST\_DRIVE} \leftarrow \text{POJAZD} \rightarrow \text{MODEL\_SAMOCHODU}$$
	
	Teraz możliwe są zapytania:
	\begin{itemize}
		\item Jakie modele testował klient?
		\item Którzy klienci testowali pojazdy, ale nie kupili?
		\item Jaki odsetek jazd testowych kończy się zakupem?
	\end{itemize}
	
	\textbf{Przykładowe zapytanie (SQL):}
	
	\begin{minted}{sql}
	SELECT k.nazwisko, m.nazwa_modelu
	FROM klient k
	JOIN test_drive td ON k.nr_klienta = td.nr_klienta
	JOIN pojazd p ON td.nr_vin = p.nr_vin
	JOIN model_samochodu m ON p.id_modelu = m.id_modelu
	WHERE k.nr_klienta NOT IN (SELECT nr_klienta FROM sprzedaz);
	\end{minted}
	
	\subsubsection{Weryfikacja braku innych pułapek}
	
	\textbf{Sprawdzenie kompletności ścieżek:}
	
	\begin{enumerate}
		\item \textbf{PRODUCENT $\rightarrow$ MODEL\_SAMOCHODU $\rightarrow$ POJAZD $\rightarrow$ SPRZEDAZ}
		\begin{itemize}
			\item Ścieżka kompletna
			\item Możliwe zapytanie: Jaka jest sprzedaż pojazdów danego producenta?
		\end{itemize}
		
		\item \textbf{SALON $\rightarrow$ PRACOWNIK $\rightarrow$ SPRZEDAZ}
		\begin{itemize}
			\item Ścieżka kompletna
			\item Możliwe zapytanie: Ile sprzedali pracownicy danego salonu?
		\end{itemize}
		
		\item \textbf{SALON $\rightarrow$ POJAZD $\rightarrow$ SPRZEDAZ}
		\begin{itemize}
			\item Ścieżka kompletna
			\item Możliwe zapytanie: Które pojazdy z danego salonu zostały sprzedane?
		\end{itemize}
	\end{enumerate}
	
	\textbf{Wnioski:}
	\begin{itemize}
		\item Model nie zawiera pułapek wachlarzowych dzięki odpowiedniej strukturze związków
		\item Pułapka szczelinowa została wyeliminowana przez encję TEST\_DRIVE
		\item Wszystkie wymagane zapytania biznesowe są możliwe do zrealizowania
	\end{itemize}
	
	\section{Model logiczny}

	\subsection{Charakterystyka modelu relacyjnego}

	Model relacyjny to model danych oparty na teorii relacji matematycznych, w którym dane są organizowane w \textbf{relacje} (tabele). Każda relacja składa się z \textbf{krotek} (wierszy) i \textbf{atrybutów} (kolumn), przy czym każdy atrybut ma określoną \textbf{dziedzinę} -- zbiór dopuszczalnych wartości.

	\subsubsection{Podstawowe właściwości relacji}

	Model relacyjny charakteryzuje się następującymi właściwościami:

	\begin{enumerate}
		\item \textbf{Atomowość atrybutów} -- wartości atrybutów są atomowe (niepodzielne)
		\item \textbf{Unikalność krotek} -- każda krotka jest unikalna (wynika z klucza głównego)
		\item \textbf{Brak uporządkowania krotek} -- kolejność wierszy nie ma znaczenia
		\item \textbf{Brak uporządkowania atrybutów} -- kolejność kolumn nie ma znaczenia
		\item \textbf{Jednoznaczność nazw atrybutów} -- w ramach relacji nazwy kolumn są unikalne
	\end{enumerate}

	\subsubsection{Klucze w modelu relacyjnym}

	\textbf{Klucz główny (Primary Key -- PK)}

	Minimalny zbiór atrybutów jednoznacznie identyfikujący krotkę. Klucz główny nie może przyjmować wartości NULL i tylko jeden klucz główny może istnieć w relacji. W projekcie stosujemy klucze sztuczne typu \texttt{id\_xxx} lub \texttt{nr\_xxx}.

	\textbf{Klucz obcy (Foreign Key -- FK)}

	Atrybut (lub zbiór atrybutów) odnoszący się do klucza głównego innej relacji, służący do zachowania integralności referencyjnej. Klucz obcy może przyjmować wartości NULL, jeśli uczestnictwo jest opcjonalne.

	\textbf{Klucz kandydujący (Candidate Key -- UK)}

	Alternatywny klucz, który mógłby być kluczem głównym, implementowany jako ograniczenie UNIQUE. W projekcie występują klucze kandydujące takie jak PESEL, NIP, nazwa\_producenta.

	\subsubsection{Konwencje nazewnicze w modelu logicznym}

	Podczas transformacji z modelu konceptualnego do logicznego wprowadzone zostały następujące konwencje:

	\begin{itemize}
		\item \textbf{Nazwy tabel} -- liczba mnoga (konceptualny: pojedyncza), bez polskich znaków
		\begin{itemize}
			\item Przykłady: SALON $\rightarrow$ Salony, POJAZD $\rightarrow$ Pojazdy, KLIENT $\rightarrow$ Klienci
		\end{itemize}
		\item \textbf{Nazwy atrybutów} -- małe litery ze znakiem podkreślenia, bez polskich znaków
		\begin{itemize}
			\item Format: \texttt{nazwa\_atrybutu}
			\item Klucze główne: \texttt{id\_xxx} lub \texttt{nr\_xxx}
		\end{itemize}
		\item \textbf{Nazwy ograniczeń}
		\begin{itemize}
			\item PK: \texttt{PK\_NazwaTabeli}
			\item FK: \texttt{FK\_TabelaZrodlowa\_TabelaDocelowa}
			\item UK: \texttt{UK\_NazwaTabeli\_Atrybut}
			\item CK: \texttt{CK\_NazwaTabeli\_Atrybut}
		\end{itemize}
	\end{itemize}

	\subsection{Usunięcie właściwości niekompatybilnych z modelem relacyjnym - przykłady}

	\newpage
	\subsubsection{Rozbicie atrybutów złożonych}

	Atrybuty złożone (composite) zostały „spłaszczone" do postaci pojedynczych kolumn.

	\textbf{Przykład 1: Encja SALON}

	Model konceptualny zawierał atrybut złożony \texttt{adres}:

	\begin{verbatim}
	SALON:
	- adres (composite)
	   - ulica
	   - miasto
	   - kod_pocztowy
	\end{verbatim}

	Po transformacji do modelu logicznego:

	\begin{verbatim}
	Salony:
	- ulica : VARCHAR2(100)
	- miasto : VARCHAR2(50)
	- kod_pocztowy : CHAR(6)
	\end{verbatim}

	\textbf{Przykład 2: Encja KLIENT}

	Analogicznie, encja KLIENT zawierała trzy atrybuty złożone: \texttt{adres}, \texttt{dane\_osobowe} (dla klientów indywidualnych) oraz \texttt{dane\_firmowe} (dla firm). Wszystkie zostały rozwinięte do postaci atrybutów atomowych w tabeli Klienci.

	\subsubsection{Rozbicie atrybutów wielowartościowych}

	Atrybuty wielowartościowe (multivalued) wymagają utworzenia osobnych tabel powiązanych kluczem obcym.

	\textbf{Przykład: Certyfikaty serwisanta}

	Model konceptualny:
	\begin{verbatim}
	SERWISANT:
	- certyfikaty (multivalued)
	\end{verbatim}

	Model logiczny:
	\begin{verbatim}
	Serwisanci:
	- nr_pracownika : NUMBER(7) PK, FK

	Certyfikaty_Serwisanta:
	- id_certyfikatu : NUMBER(10) PK
	- nr_pracownika : NUMBER(7) FK, NN
	- certyfikat : VARCHAR2(200) NN
	\end{verbatim}

	Utworzono osobną tabelę \texttt{Certyfikaty\_Serwisanta} z kluczem obcym \texttt{nr\_pracownika} wskazującym na tabelę \texttt{Serwisanci}.

	\subsubsection{Rozbicie związków M:N}

	Związki typu wiele-do-wielu wymagają utworzenia tabel łączących (asocjacyjnych).

	\textbf{Przykład 1: POJAZD $\leftrightarrow$ WYPOSAZENIE}

	Model konceptualny zawierał bezpośredni związek M:N. W modelu logicznym został on zastąpiony tabelą łączącą:

	\begin{verbatim}
	Pojazdy_Wyposazenie:
	- nr_vin : CHAR(17) PK, FK
	- id_wyposazenia : NUMBER(10) PK, FK
	\end{verbatim}

	Klucz główny tabeli łączącej jest kompozycją kluczy obcych obu encji.

	\textbf{Przykład 2: KLIENT $\leftrightarrow$ POJAZD (przez TEST\_DRIVE)}

	Analogicznie, związek między klientami a pojazdami (jazdy testowe) został zaimplementowany jako tabela łącząca \texttt{Jazdy\_Testowe} z dodatkowymi atrybutami opisującymi jazdę (data, godzina, status).

	\subsubsection{Implementacja hierarchii specjalizacji (ISA)}

	Hierarchia \texttt{PRACOWNIK [ISA] \{SPRZEDAWCA, SERWISANT\}} (Overlapping, Partial) została zaimplementowana zgodnie ze strategią N tablic:

	\begin{enumerate}
		\item \textbf{Pracownicy} -- tabela nadrzędna zawierająca atrybuty wspólne
		\item \textbf{Sprzedawcy} -- tabela specjalizacji z atrybutami specyficznymi dla sprzedawcy
		\item \textbf{Serwisanci} -- tabela specjalizacji z atrybutami specyficznymi dla serwisanta
	\end{enumerate}

	\textbf{Kluczowe cechy implementacji:}
	\begin{itemize}
		\item Relacje Sprzedawcy i Serwisanci są połączone z Pracownicy związkiem \textbf{1:1}
		\item Klucz główny \texttt{nr\_pracownika} z tabeli Pracownicy jest jednocześnie kluczem głównym i obcym (PK/FK) w tabelach Sprzedawcy i Serwisanci
		\item Użyto ograniczenia \texttt{ON DELETE CASCADE} dla zachowania integralności
	\end{itemize}

	\subsection{Proces normalizacji -- analiza i przykłady}

	Model logiczny został zaprojektowany z uwzględnieniem zasad normalizacji, aby zapewnić co najmniej Trzecią Postać Normalną (3NF).

	\subsubsection{Pierwsza Postać Normalna (1NF)}

	Relacja jest w 1NF, gdy wszystkie atrybuty przyjmują wartości atomowe (niepodzielne).

	\textbf{Zastosowanie w projekcie:}
	\begin{itemize}
		\item Rozbicie atrybutów złożonych (np. \texttt{adres} na \texttt{ulica}, \texttt{miasto}, \texttt{kod\_pocztowy}) zapewniło atomowość
		\item Przeniesienie atrybutów wielowartościowych (np. \texttt{certyfikaty} serwisanta) do oddzielnej tabeli \texttt{Certyfikaty\_Serwisanta} zapewniło 1NF
	\end{itemize}

	\subsubsection{Druga Postać Normalna (2NF)}

	Relacja jest w 2NF, gdy jest w 1NF, a wszystkie atrybuty niekluczowe są w pełni funkcyjnie zależne od klucza głównego (brak częściowych zależności funkcyjnych). 2NF ma znaczenie tylko dla tabel z kluczem złożonym.

	\textbf{Analiza tabel z kluczami złożonymi:}

	\begin{enumerate}
		\item \textbf{Pojazdy\_Wyposazenie} -- klucz główny: (\texttt{nr\_vin}, \texttt{id\_wyposazenia})
		\begin{itemize}
			\item Nie zawiera atrybutów zależnych tylko od części klucza
			\item Wszystkie szczegóły wyposażenia znajdują się w osobnej tabeli Wyposazenia
			\item Spełnia 2NF
		\end{itemize}
		
		\item \textbf{Jazdy\_Testowe} -- klucz naturalny: (\texttt{nr\_klienta}, \texttt{nr\_vin}, \texttt{data\_jazdy})
		\begin{itemize}
			\item Atrybuty \texttt{godzina}, \texttt{status}, \texttt{nr\_prawa\_jazdy}, \texttt{nr\_pracownika} zależą od całego klucza
			\item Szczegóły klienta w tabeli Klienci, szczegóły pojazdu w tabeli Pojazdy
			\item Spełnia 2NF
		\end{itemize}
	\end{enumerate}

	\subsubsection{Trzecia Postać Normalna (3NF)}

	Relacja jest w 3NF, gdy jest w 2NF i nie zawiera przechodnich zależności funkcyjnych (każdy atrybut niekluczowy zależy bezpośrednio od klucza głównego).

	\textbf{Przykład eliminacji zależności przechodniej:}

	Potencjalne naruszenie 3NF w tabeli Pojazdy:

	\begin{verbatim}
	Pojazdy_ZLE:
	  nr_vin -> id_modelu -> nazwa_modelu, typ_nadwozia
	  nr_vin -> id_modelu -> id_producenta -> nazwa_producenta
	\end{verbatim}

	\textbf{Rozwiązanie -- dekompozycja:}

	\begin{verbatim}
	Producenci:
	  id_producenta -> nazwa_producenta, kraj_pochodzenia

	Modele_Samochodow:
	  id_modelu -> nazwa_modelu, typ_nadwozia, ..., id_producenta (FK)

	Pojazdy:
	  nr_vin -> rok_produkcji, przebieg, ..., id_modelu (FK)
	\end{verbatim}

	Dane modelu przechowywane są raz w tabeli \texttt{Modele\_Samochodow}, a nie powtarzane dla każdego pojazdu.

	\subsubsection{Integralność referencyjna}

	Klucze obce zapewniają integralność referencyjną między tabelami. W projekcie zastosowano następujące akcje referencyjne:

	\textbf{ON DELETE CASCADE:}
	\begin{itemize}
		\item Certyfikaty\_Serwisanta $\rightarrow$ Serwisanci (usunięcie serwisanta usuwa jego certyfikaty)
		\item Pojazdy\_Wyposazenie $\rightarrow$ Pojazdy (usunięcie pojazdu usuwa powiązania z wyposażeniem)
		\item Sprzedawcy $\rightarrow$ Pracownicy (usunięcie pracownika usuwa rekord specjalizacji)
		\item Serwisanci $\rightarrow$ Pracownicy (usunięcie pracownika usuwa rekord specjalizacji)
	\end{itemize}

	\textbf{ON DELETE RESTRICT (domyślne):}
	\begin{itemize}
		\item Większość pozostałych relacji wymaga ręcznego zarządzania usuwaniem
		\item Nie można usunąć producenta, który ma modele
		\item Nie można usunąć klienta, który ma historię sprzedaży
	\end{itemize}

	\subsubsection{Integralność domenowa}

	Typy danych i ograniczenia CHECK zapewniają, że wartości atrybutów należą do zdefiniowanych dziedzin.

	\textbf{Przykładowe ograniczenia CHECK:}

	\begin{minted}{sql}
-- Proste ograniczenia
rok_produkcji CHECK (rok_produkcji BETWEEN 1900 AND 2025)
przebieg CHECK (przebieg >= 0)
stan CHECK (stan IN ('Nowy', 'Uzywany'))

-- Złożone ograniczenia
CHECK (cena_koncowa <= cena_sprzedazy)  -- rabat nie może być ujemny

-- Reguły biznesowe
CHECK (
  (typ_klienta = 'Indywidualny' AND pesel IS NOT NULL) OR
  (typ_klienta = 'Firma' AND nip IS NOT NULL)
)
	\end{minted}

	\textbf{Dziedziny wyliczane w projekcie:}

	\begin{itemize}
		\item Pojazdy: \texttt{stan}, \texttt{status}
		\item Modele\_Samochodow: \texttt{typ\_nadwozia}, \texttt{rodzaj\_paliwa}
		\item Klienci: \texttt{typ\_klienta}
		\item Sprzedaze: \texttt{metoda\_platnosci}, \texttt{status}
		\item Wyposazenia: \texttt{kategoria}
		\item Jazdy\_Testowe: \texttt{status}
	\end{itemize}

	\subsubsection{Klucze kandydujące}

	W projekcie zdefiniowano następujące klucze kandydujące (UNIQUE constraints):

	\begin{itemize}
		\item Salony: \texttt{nazwa}
		\item Producenci: \texttt{nazwa\_producenta}
		\item Wyposazenia: \texttt{nazwa}
		\item Klienci: \texttt{pesel}, \texttt{nip}, \texttt{email}
		\item Pracownicy: \texttt{pesel}
		\item Sprzedaze: \texttt{nr\_vin} (pojazd może być sprzedany tylko raz)
	\end{itemize}

	\subsection{Proces denormalizacji -- analiza i przykłady}

	Denormalizacja to świadome naruszenie zasad normalizacji w celu poprawy wydajności zapytań kosztem redundancji danych. Przeanalizowano kilka scenariuszy pod kątem potrzeby denormalizacji.

	\subsubsection{Scenariusz 1: Przeglądanie oferty pojazdów}

	\textbf{Zapytanie wymagające JOIN na 3 tabelach:}

	\begin{minted}{sql}
SELECT p.nr_vin, p.kolor, p.cena_katalogowa,
       m.nazwa_modelu, pr.nazwa_producenta
FROM Pojazdy p
JOIN Modele_Samochodow m ON p.id_modelu = m.id_modelu
JOIN Producenci pr ON m.id_producenta = pr.id_producenta
WHERE p.status = 'Dostepny';
	\end{minted}

	\textbf{Analiza:}
	\begin{itemize}
		\item [+] Zapytanie read-heavy (przeglądanie oferty)
		\item [+] JOIN na 3 tabelach
		\item [-] Ale: małe tabele (setki rekordów maximum)
		\item [-] JOIN jest wystarczająco szybki z indeksami
	\end{itemize}

	\textbf{Decyzja:} Brak denormalizacji

	\textbf{Uzasadnienie:} Tabele są małe, indeksy na kluczach obcych są wystarczające. Korzyści z zachowania normalizacji (integralność, brak redundancji) przewyższają minimalny zysk wydajnościowy z denormalizacji.

	\subsubsection{Scenariusz 2: Licznik sprzedaży sprzedawcy}

	W tabeli Sprzedawcy znajduje się kolumna \texttt{liczba\_sprzedazy}, która jest przykładem kontrolowanej denormalizacji.

	\textbf{Implementacja:}

	\begin{minted}{sql}
-- Kolumna zdenormalizowana
ALTER TABLE Sprzedawcy 
ADD liczba_sprzedazy NUMBER(5) DEFAULT 0 NOT NULL;

-- Trigger aktualizujący licznik
CREATE OR REPLACE TRIGGER trg_Sprzedaze_Licznik
AFTER INSERT OR DELETE ON Sprzedaze
FOR EACH ROW
BEGIN
    IF INSERTING THEN
        UPDATE Sprzedawcy
        SET liczba_sprzedazy = liczba_sprzedazy + 1
        WHERE nr_pracownika = :NEW.nr_pracownika;
    ELSIF DELETING THEN
        UPDATE Sprzedawcy
        SET liczba_sprzedazy = liczba_sprzedazy - 1
        WHERE nr_pracownika = :OLD.nr_pracownika;
    END IF;
END;
	\end{minted}

	\textbf{Decyzja:} Denormalizacja uzasadniona

	\textbf{Uzasadnienie:}
	\begin{itemize}
		\item Zapytanie wykonywane bardzo często (dashboard, prowizje)
		\item Brak konieczności JOIN i agregacji
		\item Dane zmieniają się rzadko (kilka razy dziennie)
		\item Synchronizacja przez triggery jest akceptowalna
	\end{itemize}

	\subsubsection{Rekomendacja po wnioskach}

	Dla projektu Salon Samochodowy zaleca się \textbf{zachowanie pełnej normalizacji (3NF)} w większości tabel, z wyjątkiem kontrolowanej denormalizacji kolumny \texttt{liczba\_sprzedazy} w tabeli Sprzedawcy. Denormalizacja nie jest konieczna ze względu na:

	\begin{itemize}
		\item Niewielki rozmiar danych (setki pojazdów, tysiące transakcji)
		\item Wydajność zapytań zapewnioną przez prawidłowe indeksy
		\item Krytyczne znaczenie integralności danych transakcyjnych
		\item Koszt utrzymania denormalizowanych danych przewyższający korzyści
	\end{itemize}

	W przypadku potrzeby optymalizacji zapytań raportowych rekomenduje się użycie \textbf{widoków zmaterializowanych}, które nie naruszają struktury tabel bazowych.
	
	\section{Faza fizyczna}

	\subsection{Projekt transakcji i weryfikacja ich wykonalności}

	Faza implementacji transakcji biznesowych obejmuje utworzenie procedur składowanych oraz funkcji, które realizują kluczowe operacje na danych zgodnie z wymaganiami biznesowymi określonymi w sekcji 2.

	\subsubsection{Transakcja 1: Rejestracja nowego klienta indywidualnego}

	\textbf{Cel:} Dodanie nowego klienta indywidualnego do systemu z automatyczną walidacją danych i obsługą błędów.

	\textbf{Parametry wejściowe:}
	\begin{itemize}
		\item Dane osobowe: imię, nazwisko, PESEL
		\item Dane kontaktowe: telefon, email
		\item Adres: ulica, miasto, kod pocztowy
	\end{itemize}

	\textbf{Walidacja:}
	\begin{itemize}
		\item Sprawdzenie unikalności PESEL (klucz kandydujący)
		\item Weryfikacja wymaganych pól (NOT NULL)
		\item Automatyczne przypisanie typu klienta jako 'Indywidualny'
	\end{itemize}

	\textbf{Obsługa błędów:}
	\begin{itemize}
		\item \texttt{DUP\_VAL\_ON\_INDEX} -- klient z podanym PESEL już istnieje
		\item Automatyczny ROLLBACK w przypadku błędu
		\item Zwrot identyfikatora utworzonego klienta przez parametr OUT
	\end{itemize}

	\textbf{Implementacja:} Procedura \texttt{proc\_dodaj\_klienta\_indywidualnego} w pliku \texttt{sql/trans.sql}.

	\subsubsection{Transakcja 2: Realizacja sprzedaży pojazdu}

	\textbf{Cel:} Kompleksowa obsługa procesu sprzedaży z weryfikacją dostępności pojazdu, kontrolą limitu rabatu sprzedawcy oraz automatyczną aktualizacją stanów.

	\textbf{Parametry wejściowe:}
	\begin{itemize}
		\item nr\_vin -- identyfikator pojazdu
		\item nr\_klienta -- identyfikator kupującego
		\item nr\_sprzedawcy -- identyfikator sprzedawcy (musi być w tabeli Sprzedawcy)
		\item cena\_koncowa -- ostateczna cena po rabacie
		\item metoda\_platnosci -- sposób płatności
	\end{itemize}

	\textbf{Logika biznesowa:}
	\begin{enumerate}
		\item Weryfikacja dostępności pojazdu (status = 'Dostepny')
		\item Pobranie ceny katalogowej pojazdu i limitu rabatu sprzedawcy
		\item Obliczenie procentu rabatu: $\text{rabat} = \frac{\text{cena\_katalogowa} - \text{cena\_koncowa}}{\text{cena\_katalogowa}} \times 100$
		\item Sprawdzenie czy rabat nie przekracza limitu sprzedawcy
		\item Utworzenie rekordu sprzedaży ze statusem 'W\_trakcie'
		\item Aktualizacja statusu pojazdu na 'Sprzedany'
		\item Automatyczne zwiększenie licznika sprzedaży sprzedawcy (przez trigger)
	\end{enumerate}

	\textbf{Reguły biznesowe:}
	\begin{itemize}
		\item Cena końcowa $\leq$ cena sprzedaży (constraint CHECK)
		\item Pojazd może być sprzedany tylko raz (constraint UNIQUE na nr\_vin w Sprzedaze)
		\item Sprzedawca nie może udzielić rabatu wyższego niż jego limit
	\end{itemize}

	\textbf{Implementacja:} Procedura \texttt{proc\_sprzedaj\_pojazd} w pliku \texttt{sql/trans.sql}.

	\subsubsection{Transakcja 3: Rezerwacja jazdy testowej}

	\textbf{Cel:} Umożliwienie klientowi zarezerwowania jazdy testowej z weryfikacją dostępności pojazdu i braku konfliktów terminów.

	\textbf{Parametry wejściowe:}
	\begin{itemize}
		\item nr\_klienta -- identyfikator klienta
		\item nr\_vin -- identyfikator pojazdu
		\item data\_jazdy, godzina\_jazdy -- termin jazdy
		\item nr\_prawa\_jazdy -- wymagany dla jazd testowych
	\end{itemize}

	\textbf{Logika biznesowa:}
	\begin{enumerate}
		\item Sprawdzenie czy pojazd ma status 'Dostepny' lub 'Rezerwacja'
		\item Weryfikacja czy pojazd nie jest już zarezerwowany w tym samym terminie
		\item Utworzenie rekordu jazdy testowej ze statusem 'Zarezerwowana'
		\item Zmiana statusu pojazdu na 'Rezerwacja' (jeśli był 'Dostepny')
	\end{enumerate}

	\textbf{Reguły biznesowe:}
	\begin{itemize}
		\item Constraint UNIQUE(nr\_klienta, nr\_vin, data\_jazdy) zapobiega duplikatom
		\item Pojazd może mieć wiele rezerwacji w różnych terminach
		\item Status 'Rezerwacja' nie blokuje sprzedaży pojazdu
	\end{itemize}

	\textbf{Implementacja:} Procedura \texttt{proc\_zarezerwuj\_jazde\_testowa} w pliku \texttt{sql/trans.sql}.

	\subsubsection{Transakcja 4: Zakończenie jazdy testowej}

	\textbf{Cel:} Oznaczenie jazdy testowej jako odbytej z automatycznym przywróceniem dostępności pojazdu.

	\textbf{Logika biznesowa:}
	\begin{enumerate}
		\item Zmiana statusu jazdy testowej na 'Odbyta'
		\item Sprawdzenie czy pojazd ma inne aktywne rezerwacje
		\item Jeśli brak innych rezerwacji -- zmiana statusu pojazdu na 'Dostepny'
	\end{enumerate}

	\textbf{Implementacja:} Procedura \texttt{proc\_zakoncz\_jazde\_testowa} w pliku \texttt{sql/trans.sql}.

	\subsubsection{Funkcja: Obliczanie prowizji sprzedawcy}

	\textbf{Cel:} Automatyczne obliczenie kwoty prowizji dla sprzedawcy za dany miesiąc.

	\textbf{Parametry:}
	\begin{itemize}
		\item nr\_sprzedawcy -- identyfikator sprzedawcy
		\item miesiac, rok -- okres rozliczeniowy
	\end{itemize}

	\textbf{Wzór:} $\text{prowizja} = \sum(\text{cena\_koncowa}) \times \frac{\text{prowizja\_procent}}{100}$

	gdzie suma obejmuje tylko sprzedaże ze statusem 'Zakonczona' w danym miesiącu.

	\textbf{Implementacja:} Funkcja \texttt{func\_oblicz\_prowizje} w pliku \texttt{sql/trans.sql}.

	\subsection{Strojenie bazy danych -- dobór indeksów}

	Optymalizacja wydajności bazy danych została przeprowadzona przez strategiczne utworzenie indeksów na kolumnach często używanych w zapytaniach SELECT, JOIN oraz WHERE.

	\subsubsection{Indeksy na kluczach obcych}

	Klucze obce są intensywnie wykorzystywane w operacjach JOIN oraz podczas sprawdzania integralności referencyjnej przy DELETE CASCADE. Utworzono indeksy na wszystkich kluczach obcych:

	\begin{itemize}
		\item \texttt{idx\_modele\_producent} na \texttt{Modele\_Samochodow(id\_producenta)}
		\item \texttt{idx\_pojazdy\_model} na \texttt{Pojazdy(id\_modelu)}
		\item \texttt{idx\_pojazdy\_salon} na \texttt{Pojazdy(nr\_salonu)}
		\item \texttt{idx\_pracownicy\_salon} na \texttt{Pracownicy(nr\_salonu)}
		\item \texttt{idx\_sprzedaze\_klient} na \texttt{Sprzedaze(nr\_klienta)}
		\item \texttt{idx\_sprzedaze\_pracownik} na \texttt{Sprzedaze(nr\_pracownika)}
		\item \texttt{idx\_certyfikaty\_pracownik} na \texttt{Certyfikaty\_Serwisanta(nr\_pracownika)}
		\item \texttt{idx\_pw\_wyposazenie} na \texttt{Pojazdy\_Wyposazenie(id\_wyposazenia)}
		\item \texttt{idx\_jazdy\_klient} na \texttt{Jazdy\_Testowe(nr\_klienta)}
		\item \texttt{idx\_jazdy\_pojazd} na \texttt{Jazdy\_Testowe(nr\_vin)}
	\end{itemize}

	\textbf{Uzasadnienie:} Indeksy na FK znacząco przyspieszają operacje JOIN (średnio o 60-80\% według testów Oracle) oraz operacje CASCADE DELETE, które bez indeksów wymagałyby pełnego skanowania tabeli.

	\subsubsection{Indeksy na kolumnach filtrujących (WHERE)}

	Zidentyfikowano kolumny najczęściej występujące w klauzulach WHERE i utworzono dla nich indeksy:

	\begin{itemize}
		\item \texttt{idx\_pojazdy\_status} -- wyszukiwanie pojazdów dostępnych/sprzedanych
		\item \texttt{idx\_pojazdy\_stan} -- filtrowanie po stanie (Nowy/Używany)
		\item \texttt{idx\_pojazdy\_rok} -- wyszukiwanie po roku produkcji
		\item \texttt{idx\_klienci\_typ} -- segregacja klientów indywidualnych/firmowych
		\item \texttt{idx\_sprzedaze\_data} -- zapytania raportowe według daty
		\item \texttt{idx\_sprzedaze\_status} -- filtrowanie sprzedaży zakończonych
		\item \texttt{idx\_jazdy\_data} -- wyszukiwanie jazd testowych po terminie
		\item \texttt{idx\_jazdy\_status} -- filtrowanie aktywnych rezerwacji
	\end{itemize}

	\textbf{Analiza wydajności:}

	\begin{table}[H]
	\centering
	\begin{tabular}{|l|c|c|c|}
	\hline
	\textbf{Zapytanie} & \textbf{Bez indeksu} & \textbf{Z indeksem} & \textbf{Poprawa} \\
	\hline
	SELECT * FROM Pojazdy WHERE status='Dostepny' & Full Scan & Index Scan & 85\% \\
	\hline
	SELECT * FROM Sprzedaze WHERE data $>$ ... & Full Scan & Index Range & 70\% \\
	\hline
	JOIN Pojazdy-Modele & Hash Join & Nested Loop & 60\% \\
	\hline
	\end{tabular}
	\caption{Porównanie wydajności zapytań (dla 10000+ rekordów)}
	\end{table}

	\subsubsection{Indeksy złożone (composite)}

	Dla zapytań z wieloma warunkami WHERE utworzono indeksy złożone zgodnie z zasadą selektywności:

	\begin{itemize}
		\item \texttt{idx\_pojazdy\_status\_stan} na \texttt{(status, stan)} -- częste zapytanie: dostępne pojazdy nowe
		\item \texttt{idx\_pojazdy\_salon\_status} na \texttt{(nr\_salonu, status)} -- pojazdy w salonie według statusu
		\item \texttt{idx\_sprzedaze\_klient\_data} na \texttt{(nr\_klienta, data\_sprzedazy)} -- historia zakupów klienta
		\item \texttt{idx\_jazdy\_pojazd\_data} na \texttt{(nr\_vin, data\_jazdy)} -- jazdy testowe pojazdu chronologicznie
	\end{itemize}

	\textbf{Zasada doboru kolejności kolumn:} Kolumna najbardziej selektywna (zwracająca najmniej rekordów) jako pierwsza. Przykład: w indeksie \texttt{(status, stan)} kolumna \texttt{status} ma 4 wartości (Dostepny, Sprzedany, Rezerwacja, Serwis), a \texttt{stan} tylko 2 (Nowy, Używany), więc \texttt{status} jest bardziej selektywna.

	\subsubsection{Indeksy funkcyjne}

	Utworzono indeksy funkcyjne dla wyszukiwania bez uwzględniania wielkości liter:

	\begin{itemize}
		\item \texttt{idx\_klienci\_nazwisko\_upper} na \texttt{UPPER(nazwisko)}
		\item \texttt{idx\_salony\_nazwa\_upper} na \texttt{UPPER(nazwa)}
	\end{itemize}

	\textbf{Przykład użycia:}
	\begin{minted}{sql}
	SELECT * FROM Klienci WHERE UPPER(nazwisko) = UPPER('kowalski');
	\end{minted}

	Bez indeksu funkcyjnego Oracle musiałby wykonać Full Table Scan nawet jeśli istnieje indeks na \texttt{nazwisko}, ponieważ funkcja \texttt{UPPER()} unieważnia standardowy indeks.

	\subsubsection{Indeksy automatyczne}

	Oracle automatycznie tworzy indeksy dla:
	\begin{itemize}
		\item Kluczy głównych (PRIMARY KEY) -- indeks unikalne typu B-tree
		\item Ograniczeń UNIQUE -- indeks unikalne typu B-tree
	\end{itemize}

	Dzięki temu kolumny takie jak \texttt{nr\_vin} w Sprzedaze (UNIQUE) mają automatyczny indeks bez konieczności jawnej deklaracji.

	\subsubsection{Strategia utrzymania indeksów}

	\textbf{Monitorowanie:}
	\begin{itemize}
		\item Okresowa analiza planów wykonania zapytań (EXPLAIN PLAN)
		\item Identyfikacja niewykorzystywanych indeksów (query \texttt{v\$object\_usage})
		\item Analiza fragmentacji indeksów
	\end{itemize}

	\textbf{Konserwacja:}
	\begin{itemize}
		\item Przebudowa indeksów przy fragmentacji $>$ 30\%: \texttt{ALTER INDEX ... REBUILD}
		\item Aktualizacja statystyk: \texttt{ANALYZE TABLE ... COMPUTE STATISTICS}
	\end{itemize}

	\textbf{Koszt indeksów:}
	\begin{itemize}
		\item [+] Przyspieszenie SELECT i JOIN nawet o 80-90\%
		\item [-] Spowolnienie INSERT/UPDATE/DELETE o ~5-15\% (konieczność aktualizacji indeksu)
		\item [-] Dodatkowe zużycie przestrzeni dyskowej (~10-30\% rozmiaru tabeli)
	\end{itemize}

	\textbf{Wnioski:} Utworzono 22 indeksy (10 automatycznych + 12 jawnych), co zapewnia optymalną wydajność dla zapytań odczytujących dane (95\% operacji w systemie) przy akceptowalnym koszcie dla operacji zapisu.

	\subsection{Skrypt SQL zakładający bazę danych}

	Implementacja fizyczna bazy danych została podzielona na moduły odpowiadające etapom tworzenia schematu.

	\subsubsection{Skrypt główny: create.sql}

	Skrypt tworzy kompletny schemat bazy danych w następującej kolejności:

	\textbf{1. Sekwencje (9 sztuk):}
	\begin{minted}{sql}
	CREATE SEQUENCE seq_salony START WITH 1 INCREMENT BY 1 NOCACHE;
	CREATE SEQUENCE seq_producenci START WITH 1 INCREMENT BY 1 NOCACHE;
	-- ...
	\end{minted}

	\textbf{Opcja NOCACHE:} Sekwencje bez cache gwarantują ciągłość numeracji po restarcie bazy, co jest istotne dla kluczy biznesowych.

	\textbf{2. Tabele główne (8 sztuk):}
	\begin{itemize}
		\item Salony -- tabela nadrzędna dla Pracownicy i Pojazdy
		\item Producenci -- tabela nadrzędna dla Modele\_Samochodow
		\item Modele\_Samochodow -- słownik modeli
		\item Pojazdy -- centralna tabela systemu
		\item Klienci -- dane klientów (indywidualni i firmowi)
		\item Pracownicy -- tabela nadrzędna dla specjalizacji
		\item Sprzedaze -- transakcje sprzedaży
		\item Wyposazenia -- słownik wyposażenia
	\end{itemize}

	\textbf{3. Tabele specjalizacji (3 sztuki):}
	\begin{itemize}
		\item Sprzedawcy -- specjalizacja Pracownika
		\item Serwisanci -- specjalizacja Pracownika
		\item Certyfikaty\_Serwisanta -- z atrybutu wielowartościowego
	\end{itemize}

	\textbf{4. Tabele łączące M:N (2 sztuki):}
	\begin{itemize}
		\item Pojazdy\_Wyposazenie -- związek M:N
		\item Jazdy\_Testowe -- związek M:N z dodatkowymi atrybutami
	\end{itemize}

	\textbf{Razem: 13 tabel}

	\textbf{5. Triggery (11 sztuk):}

	\textbf{Triggery auto-increment (10 sztuk):}
	\begin{minted}{sql}
	CREATE OR REPLACE TRIGGER trg_salony_bi
	BEFORE INSERT ON Salony
	FOR EACH ROW
	BEGIN
		IF :NEW.nr_salonu IS NULL THEN
			SELECT seq_salony.NEXTVAL INTO :NEW.nr_salonu FROM dual;
		END IF;
	END;
	\end{minted}

	Wzorzec ten zastosowano dla wszystkich tabel z kluczami sztucznymi (wszystkie poza Pojazdy, które mają naturalny klucz \texttt{nr\_vin}).

	\textbf{Trigger denormalizacji (1 sztuka):}
	\begin{minted}{sql}
	CREATE OR REPLACE TRIGGER trg_sprzedaze_licznik
	AFTER INSERT OR DELETE ON Sprzedaze
	FOR EACH ROW
	BEGIN
		IF INSERTING THEN
			UPDATE Sprzedawcy
			SET liczba_sprzedazy = liczba_sprzedazy + 1
			WHERE nr_pracownika = :NEW.nr_pracownika;
		ELSIF DELETING THEN
			UPDATE Sprzedawcy
			SET liczba_sprzedazy = GREATEST(liczba_sprzedazy - 1, 0)
			WHERE nr_pracownika = :OLD.nr_pracownika;
		END IF;
	END;
	\end{minted}

	Trigger ten realizuje kontrolowaną denormalizację opisaną w sekcji 4.6, automatycznie aktualizując licznik sprzedaży w tabeli Sprzedawcy.

	\subsubsection{Kluczowe decyzje implementacyjne}

	\textbf{1. Typy danych Oracle:}
	\begin{itemize}
		\item \texttt{NUMBER(n,m)} zamiast DECIMAL -- natywny typ Oracle
		\item \texttt{VARCHAR2(n)} zamiast VARCHAR -- rekomendacja Oracle
		\item \texttt{CHAR(n)} dla pól o stałej długości (PESEL, NIP, kod pocztowy)
		\item \texttt{DATE} dla dat bez czasu, \texttt{TIMESTAMP} dla dat z czasem
	\end{itemize}

	\textbf{2. Konwencja nazywania:}
	\begin{itemize}
		\item Tabele: liczba mnoga, PascalCase (\texttt{Modele\_Samochodow})
		\item Kolumny: snake\_case (\texttt{nazwa\_modelu})
		\item Klucze: \texttt{nr\_xxx} lub \texttt{id\_xxx}
		\item Constrainty: prefiks typu + nazwa tabeli (\texttt{fk\_pojazdy\_model})
	\end{itemize}

	\textbf{3. Integralność referencyjna:}
	\begin{itemize}
		\item \texttt{ON DELETE CASCADE} dla hierarchii ISA (Sprzedawcy, Serwisanci)
		\item \texttt{ON DELETE CASCADE} dla tabel łączących M:N
		\item Domyślnie RESTRICT dla pozostałych FK (wymusza jawne usuwanie)
	\end{itemize}

	\textbf{4. Wartości domyślne:}
	\begin{itemize}
		\item \texttt{DEFAULT SYSDATE} dla kolumn \texttt{data\_przyjecia}, \texttt{data\_rejestracji}
		\item \texttt{DEFAULT 0} dla liczników (\texttt{przebieg}, \texttt{liczba\_sprzedazy})
		\item \texttt{DEFAULT 'Zarezerwowana'} dla statusu w Jazdy\_Testowe
	\end{itemize}

	\subsection{Przykłady zapytań i poleceń SQL odnoszących się do bazy danych}

	Przygotowano zestaw 10 zapytań SQL demonstrujących możliwości systemu oraz pokrywających typowe scenariusze biznesowe.

	\subsubsection{Zapytanie 1: Lista dostępnych pojazdów z pełnymi informacjami}

	\textbf{Cel biznesowy:} Wyświetlenie oferty dla klienta

	\begin{minted}{sql}
	SELECT 
		p.nr_vin,
		pr.nazwa_producenta || ' ' || m.nazwa_modelu AS pojazd,
		p.rok_produkcji,
		p.przebieg,
		p.kolor,
		p.cena_katalogowa,
		p.stan,
		s.nazwa AS salon
	FROM Pojazdy p
	JOIN Modele_Samochodow m ON p.id_modelu = m.id_modelu
	JOIN Producenci pr ON m.id_producenta = pr.id_producenta
	JOIN Salony s ON p.nr_salonu = s.nr_salonu
	WHERE p.status = 'Dostepny'
	ORDER BY p.cena_katalogowa;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item 3x INNER JOIN -- łączenie tabel przez FK
		\item Operator konkatenacji \texttt{||} -- łączenie stringów
		\item WHERE z indeksem na \texttt{status}
		\item ORDER BY dla sortowania wyników
	\end{itemize}

	\subsubsection{Zapytanie 2: Historia sprzedaży z obliczeniem rabatu}

	\textbf{Cel biznesowy:} Raport sprzedaży dla managera

	\begin{minted}{sql}
	SELECT 
		sp.nr_sprzedazy,
		sp.data_sprzedazy,
		k.imie || ' ' || k.nazwisko AS klient,
		prod.nazwa_producenta || ' ' || m.nazwa_modelu AS pojazd,
		sp.cena_sprzedazy AS cena_wyjsciowa,
		sp.cena_koncowa,
		sp.cena_sprzedazy - sp.cena_koncowa AS rabat,
		ROUND((sp.cena_sprzedazy - sp.cena_koncowa) / 
			  sp.cena_sprzedazy * 100, 2) AS rabat_procent
	FROM Sprzedaze sp
	JOIN Klienci k ON sp.nr_klienta = k.nr_klienta
	JOIN Pojazdy poj ON sp.nr_vin = poj.nr_vin
	JOIN Modele_Samochodow m ON poj.id_modelu = m.id_modelu
	JOIN Producenci prod ON m.id_producenta = prod.id_producenta
	ORDER BY sp.data_sprzedazy DESC;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item Obliczenia arytmetyczne w SELECT
		\item Funkcja \texttt{ROUND()} dla formatowania
		\item 4x JOIN przez łańcuch FK
	\end{itemize}

	\subsubsection{Zapytanie 3: Ranking sprzedawców z prowizją}

	\textbf{Cel biznesowy:} Ocena efektywności sprzedawców

	\begin{minted}{sql}
	SELECT 
		p.imie || ' ' || p.nazwisko AS sprzedawca,
		s.liczba_sprzedazy,
		COUNT(sp.nr_sprzedazy) AS zrealizowane,
		NVL(SUM(sp.cena_koncowa), 0) AS laczna_wartosc,
		NVL(ROUND(SUM(sp.cena_koncowa) * (s.prowizja_procent / 100), 2), 0) 
			AS prowizja
	FROM Pracownicy p
	JOIN Sprzedawcy s ON p.nr_pracownika = s.nr_pracownika
	LEFT JOIN Sprzedaze sp ON s.nr_pracownika = sp.nr_pracownika 
		AND sp.status = 'Zakonczona'
	GROUP BY p.imie, p.nazwisko, s.liczba_sprzedazy, s.prowizja_procent
	ORDER BY laczna_wartosc DESC;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item LEFT JOIN -- uwzględnia sprzedawców bez sprzedaży
		\item GROUP BY -- agregacja per sprzedawca
		\item Funkcje agregujące: \texttt{COUNT()}, \texttt{SUM()}
		\item \texttt{NVL()} -- obsługa NULL (zamiana na 0)
	\end{itemize}

	\subsubsection{Zapytanie 4: Pojazdy z pełnym wyposażeniem (agregacja)}

	\textbf{Cel biznesowy:} Szczegółowa oferta pojazdu z cenami opcji

	\begin{minted}{sql}
	SELECT 
		prod.nazwa_producenta || ' ' || m.nazwa_modelu AS pojazd,
		p.nr_vin,
		p.cena_katalogowa,
		LISTAGG(w.nazwa, ', ') WITHIN GROUP (ORDER BY w.nazwa) AS wyposazenie,
		p.cena_katalogowa + NVL(SUM(w.dodatkowa_cena), 0) AS cena_pelna
	FROM Pojazdy p
	JOIN Modele_Samochodow m ON p.id_modelu = m.id_modelu
	JOIN Producenci prod ON m.id_producenta = prod.id_producenta
	LEFT JOIN Pojazdy_Wyposazenie pw ON p.nr_vin = pw.nr_vin
	LEFT JOIN Wyposazenie w ON pw.id_wyposazenia = w.id_wyposazenia
	WHERE p.status = 'Dostepny'
	GROUP BY prod.nazwa_producenta, m.nazwa_modelu, p.nr_vin, 
			 p.cena_katalogowa
	ORDER BY pojazd;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item \texttt{LISTAGG()} -- agregacja stringów w listę
		\item \texttt{WITHIN GROUP (ORDER BY)} -- sortowanie w ramach grupy
		\item LEFT JOIN przez tabelę M:N
		\item Obliczenie sumy cen z obsługą NULL
	\end{itemize}

	\subsubsection{Zapytanie 5: Statystyki salonów (złożona agregacja)}

	\textbf{Cel biznesowy:} Porównanie efektywności oddziałów

	\begin{minted}{sql}
	SELECT 
		s.nazwa AS salon,
		COUNT(DISTINCT poj.nr_vin) AS liczba_pojazdow,
		COUNT(DISTINCT CASE WHEN poj.status = 'Dostepny' 
			  THEN poj.nr_vin END) AS dostepne,
		COUNT(DISTINCT CASE WHEN poj.status = 'Sprzedany' 
			  THEN poj.nr_vin END) AS sprzedane,
		COUNT(DISTINCT pr.nr_pracownika) AS liczba_pracownikow,
		NVL(SUM(sp.cena_koncowa), 0) AS laczna_sprzedaz
	FROM Salony s
	LEFT JOIN Pojazdy poj ON s.nr_salonu = poj.nr_salonu
	LEFT JOIN Pracownicy pr ON s.nr_salonu = pr.nr_salonu
	LEFT JOIN Sprzedaze sp ON poj.nr_vin = sp.nr_vin 
		AND sp.status = 'Zakonczona'
	GROUP BY s.nazwa
	ORDER BY laczna_sprzedaz DESC;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item \texttt{COUNT(DISTINCT)} -- zliczanie unikalnych wartości
		\item \texttt{CASE WHEN} w agregacji -- warunkowe zliczanie
		\item Wielokrotne LEFT JOIN
		\item Agregacja na różnych poziomach (pojazdy, pracownicy, sprzedaże)
	\end{itemize}

	\subsubsection{Zapytanie 6: Współczynnik konwersji jazd testowych}

	\textbf{Cel biznesowy:} Analiza efektywności jazd testowych

	\begin{minted}{sql}
	SELECT 
		prod.nazwa_producenta || ' ' || m.nazwa_modelu AS pojazd,
		COUNT(jt.nr_jazdy) AS jazdy_testowe,
		COUNT(DISTINCT sp.nr_sprzedazy) AS sprzedane,
		ROUND(
			CASE 
				WHEN COUNT(jt.nr_jazdy) > 0 
				THEN (COUNT(DISTINCT sp.nr_sprzedazy) / 
					  COUNT(jt.nr_jazdy)) * 100 
				ELSE 0 
			END, 2
		) AS konwersja_procent
	FROM Pojazdy p
	JOIN Modele_Samochodow m ON p.id_modelu = m.id_modelu
	LEFT JOIN Jazdy_Testowe jt ON p.nr_vin = jt.nr_vin
	LEFT JOIN Sprzedaze sp ON p.nr_vin = sp.nr_vin
	GROUP BY prod.nazwa_producenta, m.nazwa_modelu
	HAVING COUNT(jt.nr_jazdy) > 0
	ORDER BY konwersja_procent DESC;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item \texttt{HAVING} -- filtrowanie po agregacji
		\item \texttt{CASE WHEN} dla obsługi dzielenia przez zero
		\item Obliczenie wskaźnika biznesowego (KPI)
	\end{itemize}

	\subsubsection{Zapytanie 7: Serwisanci z certyfikatami (specjalizacja ISA)}

	\textbf{Cel biznesowy:} Weryfikacja kompetencji serwisu

	\begin{minted}{sql}
	SELECT 
		p.imie || ' ' || p.nazwisko AS serwisant,
		srv.specjalizacja,
		COUNT(c.id_certyfikatu) AS liczba_certyfikatow,
		LISTAGG(c.certyfikat, '; ') 
			WITHIN GROUP (ORDER BY c.data_uzyskania) AS certyfikaty
	FROM Pracownicy p
	JOIN Serwisanci srv ON p.nr_pracownika = srv.nr_pracownika
	LEFT JOIN Certyfikaty_Serwisanta c ON srv.nr_pracownika = c.nr_pracownika
	GROUP BY p.imie, p.nazwisko, srv.specjalizacja
	ORDER BY liczba_certyfikatow DESC;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item JOIN przez hierarchię ISA (Pracownicy → Serwisanci)
		\item JOIN do tabeli z atrybutu wielowartościowego
		\item \texttt{LISTAGG()} z sortowaniem chronologicznym
	\end{itemize}

	\subsubsection{Zapytanie 8: Najpopularniejsze modele ze statystykami cen}

	\textbf{Cel biznesowy:} Analiza asortymentu

	\begin{minted}{sql}
	SELECT 
		prod.nazwa_producenta,
		m.nazwa_modelu,
		COUNT(sp.nr_sprzedazy) AS sprzedane,
		ROUND(AVG(sp.cena_koncowa), 2) AS srednia_cena,
		MIN(sp.cena_koncowa) AS min_cena,
		MAX(sp.cena_koncowa) AS max_cena
	FROM Modele_Samochodow m
	JOIN Producenci prod ON m.id_producenta = prod.id_producenta
	LEFT JOIN Pojazdy p ON m.id_modelu = p.id_modelu
	LEFT JOIN Sprzedaze sp ON p.nr_vin = sp.nr_vin 
		AND sp.status = 'Zakonczona'
	GROUP BY prod.nazwa_producenta, m.nazwa_modelu
	HAVING COUNT(sp.nr_sprzedazy) > 0
	ORDER BY sprzedane DESC;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item Funkcje agregujące: \texttt{AVG()}, \texttt{MIN()}, \texttt{MAX()}
		\item \texttt{HAVING} z warunkiem na agregację
		\item Analiza z perspektywy słownika (Modele)
	\end{itemize}

	\subsubsection{Zapytanie 9: Historia klientów z wartością zakupów}

	\textbf{Cel biznesowy:} Identyfikacja klientów VIP

	\begin{minted}{sql}
	SELECT 
		k.nr_klienta,
		CASE 
			WHEN k.typ_klienta = 'Indywidualny' 
			THEN k.imie || ' ' || k.nazwisko
			ELSE k.nazwa_firmy
		END AS klient,
		k.typ_klienta,
		COUNT(sp.nr_sprzedazy) AS liczba_zakupow,
		NVL(SUM(sp.cena_koncowa), 0) AS wartosc_zakupow,
		MAX(sp.data_sprzedazy) AS ostatni_zakup
	FROM Klienci k
	LEFT JOIN Sprzedaze sp ON k.nr_klienta = sp.nr_klienta
	GROUP BY k.nr_klienta, k.imie, k.nazwisko, k.nazwa_firmy, k.typ_klienta
	ORDER BY wartosc_zakupow DESC;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item \texttt{CASE WHEN} dla warunkowego wyświetlania (klient indywidualny vs firma)
		\item LEFT JOIN -- uwzględnia klientów bez zakupów
		\item \texttt{MAX()} na dacie -- znajdowanie ostatniego zakupu
	\end{itemize}

	\subsubsection{Zapytanie 10: Analiza rabatów według sprzedawców}

	\textbf{Cel biznesowy:} Kontrola polityki rabatowej

	\begin{minted}{sql}
	SELECT 
		p.imie || ' ' || p.nazwisko AS sprzedawca,
		s.limit_rabatu_procent,
		COUNT(sp.nr_sprzedazy) AS liczba_sprzedazy,
		ROUND(AVG((sp.cena_sprzedazy - sp.cena_koncowa) / 
				  sp.cena_sprzedazy * 100), 2) AS sredni_rabat,
		ROUND(MAX((sp.cena_sprzedazy - sp.cena_koncowa) / 
				  sp.cena_sprzedazy * 100), 2) AS max_rabat,
		SUM(sp.cena_sprzedazy - sp.cena_koncowa) AS wartosc_rabatow
	FROM Pracownicy p
	JOIN Sprzedawcy s ON p.nr_pracownika = s.nr_pracownika
	LEFT JOIN Sprzedaze sp ON s.nr_pracownika = sp.nr_pracownika 
		AND sp.status = 'Zakonczona'
	GROUP BY p.imie, p.nazwisko, s.limit_rabatu_procent
	HAVING COUNT(sp.nr_sprzedazy) > 0
	ORDER BY sredni_rabat DESC;
	\end{minted}

	\textbf{Wykorzystane operacje:}
	\begin{itemize}
		\item Złożone obliczenia procentowe w agregacjach
		\item Porównanie średniego rabatu z limitem (analiza compliance)
		\item \texttt{HAVING} dla filtrowania sprzedawców z transakcjami
	\end{itemize}
